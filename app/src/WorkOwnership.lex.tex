definition{
  name: validWork,
  parameters: [work : Work],
  result: Bool,
  body: ((negate ((project work owner) =? 0)) && ((negate ((project work capacity) =? 0)) && ((((project work spent) + (project work reserved)) <=? (project work capacity)) && (match (project work phase)
    | Phase.ready => (((project work spent) =? 0) && (((project work checkpoint) =? 0) && (negate ((project work remaining) =? 0))))
    | Phase.complete => (((project work remaining) =? 0) && ((project work reserved) =? 0))
    | Phase.running => true
    | Phase.paused => true))))
}

definition{
  name: initialize,
  parameters: [owner : Nat, capacity : Nat, remaining : Nat],
  result: (Result Work WorkError),
  body: (if (owner =? 0) then (ctor Result.error [Work, WorkError] (ctor WorkError.invalidOwner)) else (if ((capacity =? 0) || (remaining =? 0)) then (ctor Result.error [Work, WorkError] (ctor WorkError.invalidAmount)) else (ctor Result.ok [Work, WorkError] (record Work {owner := owner, revision := 0, capacity := capacity, spent := 0, reserved := 0, checkpoint := 0, remaining := remaining, phase := (ctor Phase.ready)}))))
}

definition{
  name: transition,
  parameters: [work : Work, owner : Nat, revision : Nat, action : Action],
  result: (Result Work WorkError),
  body: (if (negate (validWork work)) then (ctor Result.error [Work, WorkError] (ctor WorkError.invalidState)) else (if (negate (owner =? (project work owner))) then (ctor Result.error [Work, WorkError] (ctor WorkError.wrongOwner)) else (if (negate (revision =? (project work revision))) then (ctor Result.error [Work, WorkError] (ctor WorkError.staleRevision)) else (match action
    | Action.reserve amount => (match (project work phase)
      | Phase.ready => (if (amount =? 0) then (ctor Result.error [Work, WorkError] (ctor WorkError.invalidAmount)) else (if (negate ((((project work spent) + (project work reserved)) + amount) <=? (project work capacity))) then (ctor Result.error [Work, WorkError] (ctor WorkError.budgetExceeded)) else (ctor Result.ok [Work, WorkError] (record Work {owner := (project work owner), revision := ((project work revision) + 1), capacity := (project work capacity), spent := (project work spent), reserved := ((project work reserved) + amount), checkpoint := (project work checkpoint), remaining := (project work remaining), phase := (project work phase)}))))
      | Phase.paused => (if (amount =? 0) then (ctor Result.error [Work, WorkError] (ctor WorkError.invalidAmount)) else (if (negate ((((project work spent) + (project work reserved)) + amount) <=? (project work capacity))) then (ctor Result.error [Work, WorkError] (ctor WorkError.budgetExceeded)) else (ctor Result.ok [Work, WorkError] (record Work {owner := (project work owner), revision := ((project work revision) + 1), capacity := (project work capacity), spent := (project work spent), reserved := ((project work reserved) + amount), checkpoint := (project work checkpoint), remaining := (project work remaining), phase := (project work phase)}))))
      | Phase.running => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.complete => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase)))
    | Action.start => (match (project work phase)
      | Phase.ready => (if ((project work reserved) =? 0) then (ctor Result.error [Work, WorkError] (ctor WorkError.reservationMissing)) else (ctor Result.ok [Work, WorkError] (record Work {owner := (project work owner), revision := ((project work revision) + 1), capacity := (project work capacity), spent := (project work spent), reserved := (project work reserved), checkpoint := (project work checkpoint), remaining := (project work remaining), phase := (ctor Phase.running)})))
      | Phase.running => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.paused => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.complete => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase)))
    | Action.progress cost units checkpoint => (match (project work phase)
      | Phase.running => (if ((cost =? 0) || (units =? 0)) then (ctor Result.error [Work, WorkError] (ctor WorkError.invalidAmount)) else (if (negate (cost <=? (project work reserved))) then (ctor Result.error [Work, WorkError] (ctor WorkError.reservationExceeded)) else (if (negate (units <=? (project work remaining))) then (ctor Result.error [Work, WorkError] (ctor WorkError.obligationExceeded)) else (if (checkpoint <=? (project work checkpoint)) then (ctor Result.error [Work, WorkError] (ctor WorkError.checkpointNotAdvanced)) else (ctor Result.ok [Work, WorkError] (record Work {owner := (project work owner), revision := ((project work revision) + 1), capacity := (project work capacity), spent := ((project work spent) + cost), reserved := (prim subtract (project work reserved) cost : Nat), checkpoint := checkpoint, remaining := (prim subtract (project work remaining) units : Nat), phase := (project work phase)}))))))
      | Phase.ready => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.paused => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.complete => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase)))
    | Action.pause => (match (project work phase)
      | Phase.running => (ctor Result.ok [Work, WorkError] (record Work {owner := (project work owner), revision := ((project work revision) + 1), capacity := (project work capacity), spent := (project work spent), reserved := (project work reserved), checkpoint := (project work checkpoint), remaining := (project work remaining), phase := (ctor Phase.paused)}))
      | Phase.ready => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.paused => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.complete => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase)))
    | Action.resume => (match (project work phase)
      | Phase.paused => (if ((project work reserved) =? 0) then (ctor Result.error [Work, WorkError] (ctor WorkError.reservationMissing)) else (ctor Result.ok [Work, WorkError] (record Work {owner := (project work owner), revision := ((project work revision) + 1), capacity := (project work capacity), spent := (project work spent), reserved := (project work reserved), checkpoint := (project work checkpoint), remaining := (project work remaining), phase := (ctor Phase.running)})))
      | Phase.ready => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.running => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.complete => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase)))
    | Action.finish => (match (project work phase)
      | Phase.running => (if (negate ((project work remaining) =? 0)) then (ctor Result.error [Work, WorkError] (ctor WorkError.unfinishedWork)) else (ctor Result.ok [Work, WorkError] (record Work {owner := (project work owner), revision := ((project work revision) + 1), capacity := (project work capacity), spent := (project work spent), reserved := 0, checkpoint := (project work checkpoint), remaining := (project work remaining), phase := (ctor Phase.complete)})))
      | Phase.ready => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.paused => (if (negate ((project work remaining) =? 0)) then (ctor Result.error [Work, WorkError] (ctor WorkError.unfinishedWork)) else (ctor Result.ok [Work, WorkError] (record Work {owner := (project work owner), revision := ((project work revision) + 1), capacity := (project work capacity), spent := (project work spent), reserved := 0, checkpoint := (project work checkpoint), remaining := (project work remaining), phase := (ctor Phase.complete)})))
      | Phase.complete => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase)))
    | Action.transfer nextOwner => (match (project work phase)
      | Phase.paused => (if (nextOwner =? 0) then (ctor Result.error [Work, WorkError] (ctor WorkError.invalidOwner)) else (ctor Result.ok [Work, WorkError] (record Work {owner := nextOwner, revision := ((project work revision) + 1), capacity := (project work capacity), spent := (project work spent), reserved := (project work reserved), checkpoint := (project work checkpoint), remaining := (project work remaining), phase := (project work phase)})))
      | Phase.ready => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.running => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase))
      | Phase.complete => (ctor Result.error [Work, WorkError] (ctor WorkError.invalidPhase)))))))
}

definition{
  name: boundary,
  parameters: [],
  result: String,
  body: "This pure library models one owned work item. Capacity, reservation and spending are semantic units, not physical measurements. Progress records a caller-supplied monotonically advancing checkpoint and decreases outstanding units; it does not verify achieved external effects. Suspension retains reservations, spending, checkpoint and owner. Completion releases unused reservation. Callers must authenticate owner identifiers, persist transitions atomically and bind checkpoints to actual achieved state before using this model to authorize external effects. Error results leave the input state unchanged. Browser adapters expose these pure operations only through a qualified generated guest; authentication and durable persistence remain separate."
}
