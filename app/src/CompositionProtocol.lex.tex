definition{

  parameters: [words : (List Nat)],
  result: Bytes,
  body: (match (readPolicy words 0)
    | Option.none => 0x6d616c666f726d6564436f6d706f736974696f6e

      | Option.none => 0x6d616c666f726d6564436f6d706f736974696f6e

        | Result.ok accepted => 0x6163636570746564) else 0x6d616c666f726d6564436f6d706f736974696f6e)))
}

definition{
  name: boundary,
  parameters: [],
  result: String,
  body: "Composition wire v1 is compose followed by one ASCII space and lowercase hexadecimal octets of little-endian unsigned 32-bit words. Each word is eight ASCII characters so the text view preserves every request byte. Uppercase and nonhexadecimal characters refuse. Policy is CPU, memory, grants list, targets list, evidence list. Each list begins with its item count. Then comes the feature count and features: key, version, CPU, memory; vocabulary, outputs, state, events, errors, effects, conflicts, targets, accessibility and quality ID lists; dependency key/version pairs; input producer-key/version/schema triples. Maximum payload is 32768 ASCII bytes representing at most 16384 decoded bytes, 32 features, 64 entries per list. Truncation, excess counts and trailing words return malformedComposition; oversized payload returns compositionTooLarge. No semantics are decided by the browser codec. Admission errors and accepted are the generated model result. The wire bounds are not runtime resource budgets."
}
