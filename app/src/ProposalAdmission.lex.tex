definition{
  name: proposalRegistryValid,
  parameters: [registry : (List FeatureComposition::FeatureContract)],
  result: Bool,
  body: (match registry
    | List.nil => true
    | List.cons head rest => ((FeatureComposition::validFeature head) && ((if (FeatureComposition::hasRef rest (record FeatureComposition::FeatureRef {key := (project head key), version := (project head version)})) then false else true) && (proposalRegistryValid rest)))),
  recursive_argument: registry
}

definition{
  name: findProposalFeature,
  parameters: [
    registry : (List FeatureComposition::FeatureContract),
    reference : FeatureComposition::FeatureRef
  ],
  result: (Option FeatureComposition::FeatureContract),
  body: (match registry
    | List.nil => (ctor Option.none [FeatureComposition::FeatureContract])
    | List.cons head rest => (if (((project head key) =? (project reference key)) && ((project head version) =? (project reference version))) then (ctor Option.some [FeatureComposition::FeatureContract] head) else (findProposalFeature rest reference))),
  recursive_argument: registry
}

definition{
  name: resolveProposalFeatures,
  parameters: [
    registry : (List FeatureComposition::FeatureContract),
    references : (List FeatureComposition::FeatureRef)
  ],
  result: (Result (List FeatureComposition::FeatureContract) ProposalError),
  body: (match references
    | List.nil => (ctor Result.ok [(List FeatureComposition::FeatureContract), ProposalError] (nil FeatureComposition::FeatureContract))
    | List.cons head rest => (match (findProposalFeature registry head)
      | Option.none => (ctor Result.error [(List FeatureComposition::FeatureContract), ProposalError] (ctor ProposalError.unregisteredFeature))
      | Option.some feature => (match (resolveProposalFeatures registry rest)
        | Result.error error => (ctor Result.error [(List FeatureComposition::FeatureContract), ProposalError] error)
        | Result.ok remaining => (ctor Result.ok [(List FeatureComposition::FeatureContract), ProposalError] (cons feature remaining))))),
  recursive_argument: references
}

definition{
  name: admitProposal,
  parameters: [
    registry : (List FeatureComposition::FeatureContract),
    expectedIntent : Nat,
    expectedRevision : Nat,
    proposal : Proposal,
    policy : FeatureComposition::CompositionPolicy
  ],
  result: (Result (List FeatureComposition::FeatureContract) ProposalError),
  body: (if (proposalRegistryValid registry) then (if ((0 <? expectedIntent) && ((0 <? expectedRevision) && (((project proposal confidence) <=? 10000) && ((FeatureComposition::validIds (project proposal ambiguities)) && (FeatureComposition::validIds (project proposal unresolved)))))) then (if ((project proposal intent) =? expectedIntent) then (if ((project proposal registryRevision) =? expectedRevision) then (if ((prim length (project proposal ambiguities) : Nat) =? 0) then (if ((prim length (project proposal unresolved) : Nat) =? 0) then (match (resolveProposalFeatures registry (project proposal selected))
    | Result.error error => (ctor Result.error [(List FeatureComposition::FeatureContract), ProposalError] error)
    | Result.ok features => (match (FeatureComposition::admitComposition policy features)
      | Result.error error => (ctor Result.error [(List FeatureComposition::FeatureContract), ProposalError] (ctor ProposalError.compositionRejected error))
      | Result.ok accepted => (ctor Result.ok [(List FeatureComposition::FeatureContract), ProposalError] features))) else (ctor Result.error [(List FeatureComposition::FeatureContract), ProposalError] (ctor ProposalError.unresolvedConstraint))) else (ctor Result.error [(List FeatureComposition::FeatureContract), ProposalError] (ctor ProposalError.unresolvedAmbiguity))) else (ctor Result.error [(List FeatureComposition::FeatureContract), ProposalError] (ctor ProposalError.staleRegistry))) else (ctor Result.error [(List FeatureComposition::FeatureContract), ProposalError] (ctor ProposalError.staleIntent))) else (ctor Result.error [(List FeatureComposition::FeatureContract), ProposalError] (ctor ProposalError.invalidProposal))) else (ctor Result.error [(List FeatureComposition::FeatureContract), ProposalError] (ctor ProposalError.invalidRegistry)))
}

definition{
  name: proposalBoundary,
  parameters: [],
  result: String,
  body: "A router proposal supplies selected versioned references in dependency order, an intent identifier, a registry revision, unresolved ambiguity and constraint identifiers, and confidence in 0..10000. Admission resolves canonical contracts from the supplied registry rather than accepting router-authored contracts. Duplicate key/version pairs or invalid registry declarations fail; distinct versions remain separately selectable. Expected intent and registry revision must match exactly; unresolved issues refuse acceptance. Confidence does not grant permissions, evidence, budget or composition validity. The authoritative composition admission runs on every resolved selection. This pure declaration check does not authenticate the registry, establish intent interpretation, run R4, measure relevance or latency, prove evidence authenticity, generate a UI or execute effects. Router ranking and held-out qualification remain independent requirements."
}
