definition{
  name: containsId,
  parameters: [ids : (List Nat), targetId : Nat],
  result: Bool,
  body: (match ids
    | List.nil => false
    | List.cons head rest => (if (head =? targetId) then true else (containsId rest targetId))),
  recursive_argument: ids
}

definition{
  name: validIds,
  parameters: [ids : (List Nat)],
  result: Bool,
  body: (match ids
    | List.nil => true
    | List.cons head rest => (if (head =? 0) then false else (if (containsId rest head) then false else (validIds rest)))),
  recursive_argument: ids
}

definition{
  name: hasIds,
  parameters: [ids : (List Nat), required : (List Nat)],
  result: Bool,
  body: (match required
    | List.nil => true
    | List.cons head rest => ((containsId ids head) && (hasIds ids rest))),
  recursive_argument: required
}

definition{
  name: hasKey,
  parameters: [features : (List FeatureContract), key : Nat],
  result: Bool,
  body: (match features
    | List.nil => false
    | List.cons head rest => (if ((project head key) =? key) then true else (hasKey rest key))),
  recursive_argument: features
}

definition{
  name: hasRef,
  parameters: [features : (List FeatureContract), reference : FeatureRef],
  result: Bool,
  body: (match features
    | List.nil => false
    | List.cons head rest => (if (((project head key) =? (project reference key)) && ((project head version) =? (project reference version))) then true else (hasRef rest reference))),
  recursive_argument: features
}

definition{
  name: hasNeeds,
  parameters: [features : (List FeatureContract), needs : (List FeatureRef)],
  result: Bool,
  body: (match needs
    | List.nil => true
    | List.cons head rest => ((hasRef features head) && (hasNeeds features rest))),
  recursive_argument: needs
}

definition{
  name: hasInput,
  parameters: [features : (List FeatureContract), input : FeatureInput],
  result: Bool,
  body: (match features
    | List.nil => false
    | List.cons head rest => (if (((project head key) =? (project (project input producer) key)) && ((project head version) =? (project (project input producer) version))) then (containsId (project head outputs) (project input schema)) else (hasInput rest input))),
  recursive_argument: features
}

definition{
  name: hasInputs,
  parameters: [features : (List FeatureContract), inputs : (List FeatureInput)],
  result: Bool,
  body: (match inputs
    | List.nil => true
    | List.cons head rest => ((hasInput features head) && (hasInputs features rest))),
  recursive_argument: inputs
}

definition{
  name: hasConflict,
  parameters: [features : (List FeatureContract), feature : FeatureContract],
  result: Bool,
  body: (match features
    | List.nil => false
    | List.cons head rest => (if ((containsId (project feature conflicts) (project head key)) || (containsId (project head conflicts) (project feature key))) then true else (hasConflict rest feature))),
  recursive_argument: features
}

definition{
  name: validFeature,
  parameters: [feature : FeatureContract],
  result: Bool,
  body: ((0 <? (project feature key)) && ((0 <? (project feature version)) && ((match (project feature vocabulary)
    | List.nil => false
    | List.cons head rest => true) && ((match (project feature targets)
    | List.nil => false
    | List.cons head rest => true) && ((validIds (project feature vocabulary)) && ((validIds (project feature outputs)) && ((validIds (project feature state)) && ((validIds (project feature events)) && ((validIds (project feature errors)) && ((validIds (project feature effects)) && ((validIds (project feature conflicts)) && ((validIds (project feature targets)) && ((validIds (project feature accessibility)) && ((validIds (project feature quality)) && true))))))))))))))
}

definition{
  name: admitFeature,
  parameters: [
    policy : CompositionPolicy,
    accepted : (List FeatureContract),
    feature : FeatureContract,
    usedCpu : Nat,
    usedMemory : Nat
  ],
  result: (Result Bool CompositionError),
  body: (if (validFeature feature) then (if (if (hasKey accepted (project feature key)) then false else true) then (if (hasNeeds accepted (project feature needs)) then (if (hasInputs accepted (project feature inputs)) then (if (if (hasConflict accepted feature) then false else true) then (if (hasIds (project policy grants) (project feature effects)) then (if (hasIds (project policy targets) (project feature targets)) then (if ((hasIds (project policy evidence) (project feature accessibility)) && (hasIds (project policy evidence) (project feature quality))) then (if (((usedCpu + (project feature cpu)) <=? (project policy cpu)) && ((usedMemory + (project feature memory)) <=? (project policy memory))) then (ctor Result.ok [Bool, CompositionError] true) else (ctor Result.error [Bool, CompositionError] (ctor CompositionError.budgetExceeded))) else (ctor Result.error [Bool, CompositionError] (ctor CompositionError.missingEvidence))) else (ctor Result.error [Bool, CompositionError] (ctor CompositionError.unsupportedTarget))) else (ctor Result.error [Bool, CompositionError] (ctor CompositionError.permissionDenied))) else (ctor Result.error [Bool, CompositionError] (ctor CompositionError.featureConflict))) else (ctor Result.error [Bool, CompositionError] (ctor CompositionError.incompatibleInput))) else (ctor Result.error [Bool, CompositionError] (ctor CompositionError.missingDependency))) else (ctor Result.error [Bool, CompositionError] (ctor CompositionError.duplicateFeature))) else (ctor Result.error [Bool, CompositionError] (ctor CompositionError.invalidDeclaration)))
}

definition{
  name: admitRemaining,
  parameters: [
    policy : CompositionPolicy,
    accepted : (List FeatureContract),
    remaining : (List FeatureContract),
    usedCpu : Nat,
    usedMemory : Nat
  ],
  result: (Result Bool CompositionError),
  body: (match remaining
    | List.nil => (ctor Result.ok [Bool, CompositionError] true)
    | List.cons head rest => (match (admitFeature policy accepted head usedCpu usedMemory)
      | Result.error error => (ctor Result.error [Bool, CompositionError] error)
      | Result.ok admitted => (admitRemaining policy (cons head accepted) rest (usedCpu + (project head cpu)) (usedMemory + (project head memory))))),
  recursive_argument: remaining
}

definition{
  name: admitComposition,
  parameters: [policy : CompositionPolicy, features : (List FeatureContract)],
  result: (Result Bool CompositionError),
  body: (match features
    | List.nil => (ctor Result.error [Bool, CompositionError] (ctor CompositionError.emptyComposition))
    | List.cons head rest => (admitRemaining policy (nil FeatureContract) features 0 0))
}

definition{
  name: boundary,
  parameters: [],
  result: String,
  body: "Feature identifiers, schemas, effects, targets and evidence obligations are positive interned registry symbols. Admission checks declarations, not registry authenticity or the validity of evidence. Input producers and explicit dependencies must precede consumers at their exact version, so accepted ordering is acyclic. Conflicts are checked in both directions. Resource costs are additive declared upper bounds, not measured runtime work. Pure features may have empty state, events, errors, effects and outputs. The two application surfaces pass separate policies and compositions. This module does not interpret natural language or generate a UI."
}
