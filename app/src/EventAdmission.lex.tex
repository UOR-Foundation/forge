definition{
  name: eventBytesValid,
  parameters: [bytes : (List Nat)],
  result: Bool,
  body: (match bytes
    | List.nil => true
    | List.cons head rest => ((head <=? 255) && (eventBytesValid rest))),
  recursive_argument: bytes
}

definition{
  name: eventPayloadValid,
  parameters: [contract : EventContract, payload : EventPayload],
  result: Bool,
  body: (match payload
    | EventPayload.natural value => (((project contract kind) =? 0) && ((value <=? 4294967295) && (4 <=? (project contract maximumBytes))))
    | EventPayload.flag value => ((project contract kind) =? 1)
    | EventPayload.bytes values => (((project contract kind) =? 2) && ((eventBytesValid values) && ((prim length values : Nat) <=? (project contract maximumBytes)))))
}

definition{
  name: admitEvent,
  parameters: [
    features : (List FeatureComposition::FeatureContract),
    policy : FeatureComposition::CompositionPolicy,
    contract : EventContract,
    expectedGeneration : Nat,
    expectedSequence : Nat,
    envelope : EventEnvelope
  ],
  result: (Result Nat EventError),
  body: (match (FeatureComposition::admitComposition policy features)
    | Result.error error => (ctor Result.error [Nat, EventError] (ctor EventError.compositionRejected error))
    | Result.ok accepted => (if (eventContractValid features contract) then (if ((0 <? expectedGeneration) && (expectedSequence <? 4294967295)) then (if ((project envelope generation) =? expectedGeneration) then (if ((project envelope sequence) =? expectedSequence) then (if (((project envelope key) =? (project contract key)) && ((project envelope version) =? (project contract version))) then (if ((eventHasRef (project contract producers) (project envelope producer)) && (eventHasRef (project contract consumers) (project envelope consumer))) then (if (eventPayloadValid contract (project envelope payload)) then (ctor Result.ok [Nat, EventError] (expectedSequence + 1)) else (ctor Result.error [Nat, EventError] (ctor EventError.invalidPayload))) else (ctor Result.error [Nat, EventError] (ctor EventError.undeclaredRoute))) else (ctor Result.error [Nat, EventError] (ctor EventError.wrongEvent))) else (ctor Result.error [Nat, EventError] (ctor EventError.staleSequence))) else (ctor Result.error [Nat, EventError] (ctor EventError.staleGeneration))) else (ctor Result.error [Nat, EventError] (ctor EventError.invalidEventContext))) else (ctor Result.error [Nat, EventError] (ctor EventError.invalidEventContract))))
}

  name: eventAdmissionBoundary,
  parameters: [],
  result: String,
  body: "Pure admission rechecks the current declared composition policy, exact event and feature versions, declared routes, payload representation, generation and sequence. Natural payloads are uint32, flags one byte and byte arrays contain only 0..255; the declared maximum is 1..65536 bytes. A natural occupies four bytes. Sequence exhaustion refuses before uint32 increment. Feature event IDs must declare the routed event key; the event contract supplies its exact version. Acceptance returns the next sequence without mutation or delivery. The caller must authenticate declarations and participants, atomically compare and advance durable state and bind policy revisions to actual execution. This operation does not establish those facts, delivery, exactly-once effects, privacy, elapsed-time bounds or application completion. Event routes are separate from acyclic composition dependencies and may form feedback paths."
