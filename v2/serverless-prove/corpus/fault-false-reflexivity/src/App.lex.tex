definition{
  name: textOf,
  parameters: [fields : (List Foundation.View.Browser.V1.Model::FieldValue), node : Nat],

    | List.nil => ""
    | List.cons head rest => (if ((project head node) =? node) then (match (project head value)
      | Foundation.View.Browser.V1.Model::Value.Text words => words
      | Foundation.View.Browser.V1.Model::Value.Choice chosen => "") else (textOf rest node))),
  recursive_argument: fields
}

definition{
  name: accepted,
  parameters: [words : String],
  result: Bool,
  body: ((1 <=? (prim length (prim utf8_encode words : Bytes) : Nat)) && ((prim length (prim utf8_encode words : Bytes) : Nat) <=? 40))
}

definition{
  name: firstRows,
  parameters: [rows : (List Foundation.View.Browser.V1.Model::TableRow), count : Nat],
  result: (List Foundation.View.Browser.V1.Model::TableRow),
  body: (match count
    | Nat.zero => (nil Foundation.View.Browser.V1.Model::TableRow)
    | Nat.succ fewer => (match rows
      | List.nil => (nil Foundation.View.Browser.V1.Model::TableRow)
      | List.cons head rest => (cons head (firstRows rest fewer)))),
  recursive_argument: count
}

definition{
  name: added,
  parameters: [state : State, words : String],
  result: State,
  body: (if (accepted words) then (record State {revision := ((project state revision) + 1), guests := (firstRows (cons (record Foundation.View.Browser.V1.Model::TableRow {cells := (cons words (nil String))}) (project state guests)) 16)}) else state)
}

definition{
  name: cleared,
  parameters: [state : State],
  result: State,
  body: (record State {revision := ((project state revision) + 1), guests := (nil Foundation.View.Browser.V1.Model::TableRow)})
}

definition{
  name: guarded,
  parameters: [state : State, seen : Nat, pressed : Nat, words : String],
  result: State,
  body: (if (seen =? (project state revision)) then (if (pressed =? 1) then (added state words) else (if (pressed =? 2) then (cleared state) else state)) else state)
}

definition{
  name: breaksEmpty,
  parameters: [state : State, words : String],
  result: Bool,
  body: (((prim length (prim utf8_encode words : Bytes) : Nat) =? 0) && ((project state revision) <? (project (added state words) revision)))
}

definition{
  name: breaksLong,
  parameters: [state : State, words : String],
  result: Bool,
  body: ((41 <=? (prim length (prim utf8_encode words : Bytes) : Nat)) && ((project state revision) <? (project (added state words) revision)))
}

definition{
  name: breaksStale,
  parameters: [state : State, seen : Nat, pressed : Nat, words : String],
  result: Bool,
  body: (((seen <? (project state revision)) || ((project state revision) <? seen)) && (((project (guarded state seen pressed words) revision) <? (project state revision)) || ((project state revision) <? (project (guarded state seen pressed words) revision))))
}
