\begin{lexlean}{Verdict}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\title{Boolean}
\begin{semanticmodule}
structure{
  name: Evidence,
  type_parameters: [],
  parameters: [],
  fields: [{name: hasProof, type: Bool}, {name: computedKey, type: Nat}, {name: recordKey, type: Nat}, {name: bundleNamed, type: Nat}, {name: bundleDerived, type: Nat}, {name: libraryPinned, type: Nat}, {name: libraryDerived, type: Nat}, {name: specStatement, type: Nat}, {name: checkedStatement, type: Nat}, {name: typesMatch, type: Bool}, {name: allBound, type: Bool}, {name: kernelAccepted, type: Bool}, {name: axiomsObserved, type: Nat}, {name: axiomsDeclared, type: Nat}, {name: leanAttested, type: Bool}, {name: attestedBytes, type: Nat}, {name: assembledBytes, type: Nat}]
}

inductive{
  name: Verdict,
  type_parameters: [],
  parameters: [],
  constructors: [{name: proved, fields: []}, {name: kernelChecked, fields: []}, {name: pending, fields: []}, {name: refused, fields: []}]
}

definition{
  name: codeProved,
  parameters: [],
  result: Nat,
  body: 0
}

definition{
  name: codeKernelChecked,
  parameters: [],
  result: Nat,
  body: 1
}

definition{
  name: codePending,
  parameters: [],
  result: Nat,
  body: 2
}

definition{
  name: codeRefused,
  parameters: [],
  result: Nat,
  body: 3
}

definition{
  name: verdictCode,
  parameters: [e : Evidence],
  result: Nat,
  body: (if (not (project e hasProof)) then (codePending) else (if (not ((project e recordKey) =? (project e computedKey))) then (codeRefused) else (if (not ((project e bundleDerived) =? (project e bundleNamed))) then (codeRefused) else (if (not ((project e libraryDerived) =? (project e libraryPinned))) then (codeRefused) else (if (not ((project e checkedStatement) =? (project e specStatement))) then (codeRefused) else (if (not (project e typesMatch)) then (codeRefused) else (if (not (project e allBound)) then (codeRefused) else (if (not (project e kernelAccepted)) then (codeRefused) else (if (not ((project e axiomsObserved) =? (project e axiomsDeclared))) then (codeRefused) else (if ((project e leanAttested) && ((project e attestedBytes) =? (project e assembledBytes))) then (codeProved) else (codeKernelChecked)))))))))))
}

definition{
  name: verdictOf,
  parameters: [code : Nat],
  result: Verdict,
  body: (if (code =? (codeProved)) then (ctor Verdict.proved) else (if (code =? (codeKernelChecked)) then (ctor Verdict.kernelChecked) else (if (code =? (codePending)) then (ctor Verdict.pending) else (ctor Verdict.refused))))
}

definition{
  name: accepted,
  parameters: [code : Nat],
  result: Bool,
  body: ((code =? (codeProved)) || (code =? (codeKernelChecked)))
}

definition{
  name: wordProved,
  parameters: [],
  result: Nat,
  body: 0
}

definition{
  name: wordKernelChecked,
  parameters: [],
  result: Nat,
  body: 1
}

definition{
  name: wordProving,
  parameters: [],
  result: Nat,
  body: 2
}

definition{
  name: wordRefused,
  parameters: [],
  result: Nat,
  body: 3
}

definition{
  name: wordOf,
  parameters: [code : Nat],
  result: Nat,
  body: (if (code =? (codeProved)) then (wordProved) else (if (code =? (codeKernelChecked)) then (wordKernelChecked) else (if (code =? (codePending)) then (wordProving) else (wordRefused))))
}

definition{
  name: breaksStatement,
  parameters: [e : Evidence],
  result: Bool,
  body: ((accepted (verdictCode e)) && (not ((project e checkedStatement) =? (project e specStatement))))
}

theorem{
  name: acceptedOnlyForTheSpecStatement,
  parameters: [e : Evidence],
  statement: ((breaksStatement e) = false),
  proof: linear_arithmetic{definitions: [accepted, breaksStatement, codeKernelChecked, codePending, codeProved, codeRefused, verdictCode]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksProvedStatement,
  parameters: [e : Evidence],
  result: Bool,
  body: (((verdictCode e) =? (codeProved)) && (not ((project e checkedStatement) =? (project e specStatement))))
}

theorem{
  name: provedOnlyForTheSpecStatement,
  parameters: [e : Evidence],
  statement: ((breaksProvedStatement e) = false),
  proof: linear_arithmetic{definitions: [breaksProvedStatement, codeKernelChecked, codePending, codeProved, codeRefused, verdictCode]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksKey,
  parameters: [e : Evidence],
  result: Bool,
  body: ((accepted (verdictCode e)) && (not ((project e recordKey) =? (project e computedKey))))
}

theorem{
  name: noAcceptanceUnderAnotherKey,
  parameters: [e : Evidence],
  statement: ((breaksKey e) = false),
  proof: linear_arithmetic{definitions: [accepted, breaksKey, codeKernelChecked, codePending, codeProved, codeRefused, verdictCode]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksBytes,
  parameters: [e : Evidence],
  result: Bool,
  body: ((accepted (verdictCode e)) && (not ((project e bundleDerived) =? (project e bundleNamed))))
}

theorem{
  name: noAcceptanceOfAlteredBytes,
  parameters: [e : Evidence],
  statement: ((breaksBytes e) = false),
  proof: linear_arithmetic{definitions: [accepted, breaksBytes, codeKernelChecked, codePending, codeProved, codeRefused, verdictCode]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksLibrary,
  parameters: [e : Evidence],
  result: Bool,
  body: ((accepted (verdictCode e)) && (not ((project e libraryDerived) =? (project e libraryPinned))))
}

theorem{
  name: noAcceptanceOverAnotherLibrary,
  parameters: [e : Evidence],
  statement: ((breaksLibrary e) = false),
  proof: linear_arithmetic{definitions: [accepted, breaksLibrary, codeKernelChecked, codePending, codeProved, codeRefused, verdictCode]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksKernel,
  parameters: [e : Evidence],
  result: Bool,
  body: ((accepted (verdictCode e)) && (not (((project e kernelAccepted) && (project e typesMatch)) && (project e allBound))))
}

theorem{
  name: noAcceptanceWithoutTheKernel,
  parameters: [e : Evidence],
  statement: ((breaksKernel e) = false),
  proof: linear_arithmetic{definitions: [accepted, breaksKernel, codeKernelChecked, codePending, codeProved, codeRefused, verdictCode]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksAxioms,
  parameters: [e : Evidence],
  result: Bool,
  body: ((accepted (verdictCode e)) && (not ((project e axiomsObserved) =? (project e axiomsDeclared))))
}

theorem{
  name: noAcceptanceWithOtherAxioms,
  parameters: [e : Evidence],
  statement: ((breaksAxioms e) = false),
  proof: linear_arithmetic{definitions: [accepted, breaksAxioms, codeKernelChecked, codePending, codeProved, codeRefused, verdictCode]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksLean,
  parameters: [e : Evidence],
  result: Bool,
  body: (((verdictCode e) =? (codeProved)) && (not ((project e leanAttested) && ((project e attestedBytes) =? (project e assembledBytes)))))
}

theorem{
  name: provedNeedsLeanOnTheseBytes,
  parameters: [e : Evidence],
  statement: ((breaksLean e) = false),
  proof: linear_arithmetic{definitions: [breaksLean, codeKernelChecked, codePending, codeProved, codeRefused, verdictCode]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksPending,
  parameters: [e : Evidence],
  result: Bool,
  body: ((not (project e hasProof)) && (not ((verdictCode e) =? (codePending))))
}

theorem{
  name: noProofIsPending,
  parameters: [e : Evidence],
  statement: ((breaksPending e) = false),
  proof: linear_arithmetic{definitions: [breaksPending, codeKernelChecked, codePending, codeProved, codeRefused, verdictCode]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksWord,
  parameters: [code : Nat],
  result: Bool,
  body: (((wordOf code) =? (wordProved)) && (not (code =? (codeProved))))
}

theorem{
  name: onlyProvedReadsProved,
  parameters: [code : Nat],
  statement: ((breaksWord code) = false),
  proof: linear_arithmetic{definitions: [breaksWord, codeKernelChecked, codePending, codeProved, codeRefused, wordKernelChecked, wordOf, wordProved, wordProving, wordRefused]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksWeaker,
  parameters: [e : Evidence],
  result: Bool,
  body: (((verdictCode e) =? (codeKernelChecked)) && ((wordOf (verdictCode e)) =? (wordProved)))
}

theorem{
  name: kernelCheckedNeverReadsProved,
  parameters: [e : Evidence],
  statement: ((breaksWeaker e) = false),
  proof: linear_arithmetic{definitions: [breaksWeaker, codeKernelChecked, codePending, codeProved, codeRefused, verdictCode, wordKernelChecked, wordOf, wordProved, wordProving, wordRefused]},
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
