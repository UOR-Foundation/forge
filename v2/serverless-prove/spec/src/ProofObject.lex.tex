definition{
  name: admitCode,
  parameters: [wanted : Key, theorem : Nat, entry : ProofRecord, derived : Nat],
  result: Nat,
  body: (if (not ((project (project entry key) toolchain) =? (project wanted toolchain))) then (codeToolchain) else (if (not ((project (project entry key) declaration) =? (project wanted closure))) then (codeClosure) else (if (not ((project entry theorem) =? theorem)) then (codeTheorem) else (if (not ((project entry bundle) =? derived)) then (codeBytes) else (codeAdmitted))))))
