\begin{lexlean}{Reuse}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\title{Boolean}
\begin{semanticmodule}
structure{
  name: Stored,
  type_parameters: [],
  parameters: [],
  fields: [{name: key, type: Nat}, {name: address, type: Nat}, {name: verdict, type: Nat}]
}

definition{
  name: reused,
  parameters: [stored : (Option Stored), key : Nat, derived : Nat, fresh : Nat],
  result: Nat,
  body: (match stored
    | Option.none => fresh
    | Option.some entry => (if (((project entry key) =? key) && ((project entry address) =? derived)) then (project entry verdict) else fresh))
}

definition{
  name: deterministic,
  parameters: [collisionFree : Bool, entry : Stored, key : Nat, derived : Nat, fresh : Nat],
  result: Bool,
  body: ((not ((collisionFree && ((project entry key) =? key)) && ((project entry address) =? derived))) || ((project entry verdict) =? fresh))
}

definition{
  name: breaksRead,
  parameters: [entry : Stored, key : Nat, derived : Nat, fresh : Nat],
  result: Bool,
  body: ((not ((project entry address) =? derived)) && (not ((reused (ctor Option.some [Stored] entry) key derived fresh) =? fresh)))
}

theorem{
  name: storedResultFailingItsCheckIsNeverBelieved,
  parameters: [entry : Stored, key : Nat, derived : Nat, fresh : Nat],
  statement: ((breaksRead entry key derived fresh) = false),
  proof: linear_arithmetic{definitions: [breaksRead, reused]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksKey,
  parameters: [entry : Stored, key : Nat, derived : Nat, fresh : Nat],
  result: Bool,
  body: ((not ((project entry key) =? key)) && (not ((reused (ctor Option.some [Stored] entry) key derived fresh) =? fresh)))
}

theorem{
  name: storedUnderAnotherKeyIsNeverBelieved,
  parameters: [entry : Stored, key : Nat, derived : Nat, fresh : Nat],
  statement: ((breaksKey entry key derived fresh) = false),
  proof: linear_arithmetic{definitions: [breaksKey, reused]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksReuse,
  parameters: [collisionFree : Bool, stored : (Option Stored), key : Nat, derived : Nat, fresh : Nat],
  result: Bool,
  body: (match stored
    | Option.none => (not ((reused stored key derived fresh) =? fresh))
    | Option.some entry => ((collisionFree && (deterministic collisionFree entry key derived fresh)) && (not ((reused stored key derived fresh) =? fresh))))
}

theorem{
  name: reuseNeverChangesTheVerdict,
  parameters: [collisionFree : Bool, stored : (Option Stored), key : Nat, derived : Nat, fresh : Nat],
  statement: ((breaksReuse collisionFree stored key derived fresh) = false),
  proof: linear_arithmetic{definitions: [breaksReuse, deterministic, reused]},
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
