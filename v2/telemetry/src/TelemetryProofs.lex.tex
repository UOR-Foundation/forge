definition{
  name: breaksNoOver,
  parameters: [observation : Telemetry::Observation, stages : Telemetry::Stages],
  result: Bool,
  body: (((project observation over) =? 0) && (not ((Telemetry::lever observation stages) =? 0)))
}

theorem{
  name: noSuggestionWithoutOver,

  axioms: [Quot.sound, propext]
}
