definition{
  name: firstAt,
  parameters: [events : (List Event), kind : Nat, after : Nat],
  result: (Option Nat),
  body: (match events
    | List.nil => (ctor Option.none [Nat])
    | List.cons head rest => (if (((project head kind) =? kind) && (after <=? (project head at))) then (ctor Option.some [Nat] (project head at)) else (firstAt rest kind after))),
  recursive_argument: events
}

definition{
  name: span,
  parameters: [events : (List Event), start : Nat, finish : Nat],
  result: (Option Nat),
  body: (match (firstAt events start 0)
    | Option.none => (ctor Option.none [Nat])
    | Option.some t => (match (firstAt events finish t)
      | Option.none => (ctor Option.none [Nat])
      | Option.some u => (ctor Option.some [Nat] (prim subtract u t : Nat))))
}

definition{
  name: countKind,
  parameters: [events : (List Event), kind : Nat],
  result: Nat,
  body: (match events
    | List.nil => 0
    | List.cons head rest => (if ((project head kind) =? kind) then ((countKind rest kind) + 1) else (countKind rest kind))),
  recursive_argument: events
}

definition{
  name: countValue,
  parameters: [events : (List Event), kind : Nat, value : Nat],
  result: Nat,
  body: (match events
    | List.nil => 0
    | List.cons head rest => (if (((project head kind) =? kind) && ((project head value) =? value)) then ((countValue rest kind value) + 1) else (countValue rest kind value))),
  recursive_argument: events
}

definition{
  name: maxValue,
  parameters: [events : (List Event), kind : Nat, best : Nat],
  result: Nat,
  body: (match events
    | List.nil => best

  recursive_argument: events
}

definition{
  name: editWorst,
  parameters: [events : (List Event), pending : (Option Nat), worst : Nat],
  result: Nat,
  body: (match events
    | List.nil => worst
    | List.cons head rest => (if ((project head kind) =? (kindEdit)) then (editWorst rest (ctor Option.some [Nat] (project head at)) worst) else (if ((project head kind) =? (kindEditScreen)) then (match pending
      | Option.none => (editWorst rest (ctor Option.none [Nat]) worst)
      | Option.some t => (editWorst rest (ctor Option.none [Nat]) (if (worst <? (prim subtract (project head at) t : Nat)) then (prim subtract (project head at) t : Nat) else worst))) else (editWorst rest pending worst)))),
  recursive_argument: events
}

}

(kpiEdit)) then (present events (kindEditScreen) (editWorst events (ctor Option.none [Nat]) 0)) else (if (code =? (kpiChecked)) then (span events (kindComplete) (kindChecked)) else (if (code =? (kpiBuilt)) then (span events (kindEnter) (kindChecked)) else (if (code =? (kpiReloads)) then (present events (kindEnter) (countKind events (kindReload))) else (if (code =? (kpiThink)) then (span events (kindEnter) (kindToken)) else (if (code =? (kpiLead)) then (span events (kindToken) (kindDeclaration)) else (if (code =? (kpiWrite)) then (span events (kindDeclaration) (kindComplete)) else (if (code =? (kpiFirstTry)) then (present events (kindEnter) (countValue events (kindRead) 1)) else (if (code =? (kpiRechecked)) then (present events (kindEnter) (countKind events (kindRechecked))) else (ctor Option.none [Nat])))))))))))))))))

definition{
  name: countOver,
  parameters: [values : (List Nat), code : Nat, budget : Nat],
  result: Nat,
  body: (match values
    | List.nil => 0
    | List.cons head rest => ((countOver rest code budget) + (overCode code head budget))),
  recursive_argument: values
}

definition{
  name: worstOf,
  parameters: [values : (List Nat), best : Nat],
  result: Nat,
  body: (match values
    | List.nil => best
    | List.cons head rest => (worstOf rest (if (best <? head) then head else best))),
  recursive_argument: values
}

definition{
  name: totalOf,
  parameters: [values : (List Nat)],
  result: Nat,
  body: (match values | List.nil => 0 | List.cons head rest => (head + (totalOf rest))),
  recursive_argument: values
}

definition{
  name: lastOf,
  parameters: [values : (List Nat)],
  result: Nat,
  body: (match values
    | List.nil => 0
    | List.cons head rest => (match rest
      | List.nil => head
      | List.cons next more => (lastOf rest))),
  recursive_argument: values
}

(kpiPixel)) then (leverProjectEarly) else (if ((project observation kpi) =? (kpiUpdate)) then (leverCoalesce) else (if ((project observation kpi) =? (kpiEdit)) then (leverEditPath) else (if ((project observation kpi) =? (kpiChecked)) then (leverReuseChecks) else (if ((project observation kpi) =? (kpiBuilt)) then (largestStage stages) else (if ((project observation kpi) =? (kpiReloads)) then (leverReloadTrigger) else (leverNone))))))))) else (leverNone))

definition{
  name: bestSwitch,
  parameters: [
    current : ModelStat,
    others : (List ModelStat),
    built : Observation,
    best : Nat,
    bestRate : Nat
  ],
  result: Nat,
  body: (match others
    | List.nil => best
    | List.cons head rest => (if ((0 <? (switchTo current head built)) && (bestRate <? (firstTryPermille head))) then (bestSwitch current rest built (switchTo current head built) (firstTryPermille head)) else (bestSwitch current rest built best bestRate))),
  recursive_argument: others
}

definition{
  name: loopApplied,
  parameters: [loop : Loop, event : Nat],
  result: Nat,
  body: (if ((event =? (loopApprove)) && (not ((project loop pending) =? 0))) then (project loop pending) else (project loop applied))
}

definition{
  name: loopPending,
  parameters: [loop : Loop, event : Nat, offered : Nat],
  result: Nat,
  body: (if (event =? (loopOffer)) then (if (offered =? 0) then (project loop pending) else offered) else (if ((event =? (loopApprove)) || (event =? (loopDismiss))) then 0 else (project loop pending)))
}

definition{
  name: loopBaseline,
  parameters: [loop : Loop, event : Nat, offered : Nat, mean : Nat],
  result: Nat,
  body: (if ((event =? (loopOffer)) && (not (offered =? 0))) then mean else (project loop baseline))
}

definition{
  name: loopStep,
