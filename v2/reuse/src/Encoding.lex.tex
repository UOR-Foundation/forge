definition{
  name: bytesFixed,
  parameters: [value : Nat, width : Nat],
  result: Bytes,
  body: (match width
    | Nat.zero => 0x
    | Nat.succ rest => (prim append (bytesFixed (prim quotient value 256 0 : Nat) rest) (byteOf value) : Bytes)),
  recursive_argument: width
}

definition{
  name: addressBytes,
  parameters: [value : Nat],
  result: Bytes,
  body: (bytesFixed value 32)
}

  name: countBytes,
  parameters: [value : Nat],
  result: Bytes,
  body: (bytesFixed value 8)

definition{
  name: framed,
  parameters: [bytes : Bytes],
  result: Bytes,
  body: (prim append (countBytes (prim length bytes : Nat)) bytes : Bytes)
}

definition{
  name: addresses,
  parameters: [list : (List Nat)],
  result: Bytes,
  body: (match list
    | List.nil => 0x
    | List.cons head rest => (prim append (addressBytes head) (addresses rest) : Bytes)),
  recursive_argument: list
}

definition{
  name: insertOrdered,
  parameters: [value : Nat, list : (List Nat)],
  result: (List Nat),
  body: (match list
    | List.nil => (cons value (nil Nat))
    | List.cons head rest => (cons (if (value <=? head) then value else head) (insertOrdered (if (value <=? head) then head else value) rest))),
  recursive_argument: list
}

definition{
  name: ordered,
  parameters: [list : (List Nat)],
  result: (List Nat),
  body: (match list
    | List.nil => (nil Nat)
    | List.cons head rest => (insertOrdered head (ordered rest))),
  recursive_argument: list
}

definition{
  name: ascendingCode,
  parameters: [list : (List Nat)],
  result: Nat,
  body: (match list
    | List.nil => 0
    | List.cons head rest => (match rest
      | List.nil => 0
      | List.cons next more => (if (head <? next) then (ascendingCode rest) else 1))),
  recursive_argument: list
}

definition{
  name: sameList,
  parameters: [left : (List Nat), right : (List Nat)],
  result: Bool,
  body: (match left
    | List.nil => (match right
      | List.nil => true
      | List.cons head rest => false)
    | List.cons head rest => (match right
      | List.nil => false
      | List.cons other more => ((head =? other) && (sameList rest more)))),
  recursive_argument: left
}
