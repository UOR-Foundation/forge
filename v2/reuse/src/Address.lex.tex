  name: Staleness,

  parameters: [],
  constructors: [
    {name: fresh, fields: []},

definition{
  name: dependenciesValid,
  parameters: [dependencies : (List Nat)],

definition{
  name: keyPreimage,
  parameters: [key : Key],
  result: Bytes,

definition{
  name: codeFresh,
  parameters: [],
  result: Nat,
  body: 0
}

definition{
  name: codeToolchainChanged,
  parameters: [],
  result: Nat,
  body: 1
}

definition{
  name: codeOwnChanged,
  parameters: [],
  result: Nat,
  body: 2
}

definition{
  name: codeDependencyChanged,
  parameters: [],
  result: Nat,
  body: 3
}

definition{
  name: codeAdded,
  parameters: [],
  result: Nat,
  body: 4
}

definition{
  name: stalenessCode,
  parameters: [previous : Key, current : Key],
  result: Nat,
  body: (if (not ((project previous toolchain) =? (project current toolchain))) then (codeToolchainChanged) else (if (not ((project previous own) =? (project current own))) then (codeOwnChanged) else (if (not ((project previous dependencies) =? (project current dependencies))) then (codeDependencyChanged) else (codeFresh))))
}

definition{
  name: changeCode,
  parameters: [change : Change],
  result: Nat,
  body: (match (project change previous)
    | Option.none => (codeAdded)
    | Option.some previous => (stalenessCode previous (project change current)))
}

definition{
  name: stalenessOf,
  parameters: [code : Nat],
  result: Staleness,
  body: (if (code =? (codeFresh)) then (ctor Staleness.fresh) else (if (code =? (codeToolchainChanged)) then (ctor Staleness.toolchainChanged) else (if (code =? (codeOwnChanged)) then (ctor Staleness.ownChanged) else (if (code =? (codeDependencyChanged)) then (ctor Staleness.dependencyChanged) else (ctor Staleness.added)))))
}

definition{
  name: staleness,
  parameters: [change : Change],
  result: Staleness,
  body: (stalenessOf (changeCode change))
}

definition{
  name: recheck,
  parameters: [changes : (List Change)],
  result: (List Nat),
  body: (match changes
    | List.nil => (nil Nat)
    | List.cons head rest => (if ((changeCode head) =? (codeFresh)) then (recheck rest) else (cons (project head index) (recheck rest)))),
  recursive_argument: changes
}

definition{
  name: breaksUnchanged,
  parameters: [previous : Key, current : Key],
  result: Bool,
  body: (((((project previous toolchain) =? (project current toolchain)) && ((project previous own) =? (project current own))) && ((project previous dependencies) =? (project current dependencies))) && (not ((stalenessCode previous current) =? (codeFresh))))
}

theorem{
  name: unchangedKeepsAddress,
  parameters: [previous : Key, current : Key],
  statement: ((breaksUnchanged previous current) = false),
  proof: linear_arithmetic{definitions: [breaksUnchanged, codeDependencyChanged, codeFresh, codeOwnChanged, codeToolchainChanged, stalenessCode]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksToolchainChange,
  parameters: [previous : Key, current : Key],
  result: Bool,
  body: ((not ((project previous toolchain) =? (project current toolchain))) && (not ((stalenessCode previous current) =? (codeToolchainChanged))))
}

theorem{
  name: nothingReusedAcrossToolchains,
  parameters: [previous : Key, current : Key],
  statement: ((breaksToolchainChange previous current) = false),
  proof: linear_arithmetic{definitions: [breaksToolchainChange, codeDependencyChanged, codeFresh, codeOwnChanged, codeToolchainChanged, stalenessCode]},
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: breaksOwnChange,
  parameters: [previous : Key, current : Key],
  result: Bool,
  body: ((not ((project previous own) =? (project current own))) && ((stalenessCode previous current) =? (codeFresh)))
}

theorem{
  name: changedBytesAreStale,
  parameters: [previous : Key, current : Key],
  statement: ((breaksOwnChange previous current) = false),
  proof: linear_arithmetic{definitions: [breaksOwnChange, codeDependencyChanged, codeFresh, codeOwnChanged, codeToolchainChanged, stalenessCode]},
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: breaksDependencyChange,
  parameters: [previous : Key, current : Key],
  result: Bool,
  body: ((not ((project previous dependencies) =? (project current dependencies))) && ((stalenessCode previous current) =? (codeFresh)))
}

theorem{
  name: changedDependencyIsStale,
  parameters: [previous : Key, current : Key],
  statement: ((breaksDependencyChange previous current) = false),
  proof: linear_arithmetic{definitions: [breaksDependencyChange, codeDependencyChanged, codeFresh, codeOwnChanged, codeToolchainChanged, stalenessCode]},
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: breaksFreshOnlyUnchanged,
  parameters: [previous : Key, current : Key],
  result: Bool,
  body: (((stalenessCode previous current) =? (codeFresh)) && (not ((((project previous toolchain) =? (project current toolchain)) && ((project previous own) =? (project current own))) && ((project previous dependencies) =? (project current dependencies)))))
}

theorem{
  name: freshExactlyWhenUnchanged,
  parameters: [previous : Key, current : Key],
  statement: ((breaksFreshOnlyUnchanged previous current) = false),
  proof: linear_arithmetic{definitions: [breaksFreshOnlyUnchanged, codeDependencyChanged, codeFresh, codeOwnChanged, codeToolchainChanged, stalenessCode]},
  axioms: [Classical.choice, Quot.sound, propext]
}

theorem{
  name: addedIsChecked,
  parameters: [index : Nat, current : Key],
  statement: ((changeCode (record Change {index := index, previous := (ctor Option.none [Key]), current := current})) = (codeAdded)),
  proof: linear_arithmetic{definitions: [changeCode, codeAdded]},
  axioms: [propext]
}
