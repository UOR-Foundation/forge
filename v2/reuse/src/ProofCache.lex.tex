definition{
  name: admissionCode,
  parameters: [query : Query, entry : ReuseRecord, observed : (Option Nat)],
  result: Nat,
  body: (if (not ((project query address) =? (project entry address))) then (codeAddressMismatch) else (if (not ((project query toolchain) =? (project entry toolchain))) then (codeToolchainMismatch) else (if (not ((project query claim) =? (project entry claim))) then (codeClaimMismatch) else (match (project entry attestation)
    | Option.none => (codeAttestationMissing)
    | Option.some attestation => (match observed
      | Option.none => (codeAttestationUnverified)
      | Option.some seen => (if (attestation =? seen) then (codeAdmitted) else (codeAttestationUnverified)))))))
}

definition{
  name: refusalOf,
  parameters: [code : Nat],
  result: Refusal,
  body: (if (code =? (codeAddressMismatch)) then (ctor Refusal.addressMismatch) else (if (code =? (codeToolchainMismatch)) then (ctor Refusal.toolchainMismatch) else (if (code =? (codeClaimMismatch)) then (ctor Refusal.claimMismatch) else (if (code =? (codeAttestationMissing)) then (ctor Refusal.attestationMissing) else (ctor Refusal.attestationUnverified)))))
}

definition{
  name: admitRecord,
  parameters: [query : Query, entry : ReuseRecord, observed : (Option Nat)],
  result: Admission,
  body: (if ((admissionCode query entry observed) =? (codeAdmitted)) then (match (project entry attestation)
    | Option.none => (ctor Admission.refused (ctor Refusal.attestationMissing))
    | Option.some attestation => (ctor Admission.admitted attestation)) else (ctor Admission.refused (refusalOf (admissionCode query entry observed))))
}

}

definition{
  name: nextStep,
  parameters: [consulted : Nat, query : Query, stored : (Option ReuseRecord), observed : (Option Nat)],
  result: Step,
  body: (match stored
    | Option.none => (if (consulted =? 1) then (ctor Step.work) else (ctor Step.consult))
    | Option.some found => (if ((nextStepCode consulted (admissionCode query found observed)) =? (stepReuse)) then (match (admitRecord query found observed)
      | Admission.admitted attestation => (ctor Step.reuse attestation)
      | Admission.refused reason => (ctor Step.work)) else (if ((nextStepCode consulted (admissionCode query found observed)) =? (stepWork)) then (ctor Step.work) else (ctor Step.consult))))
}
