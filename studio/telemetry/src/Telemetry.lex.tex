(kpiPixel)) then (leverProjectEarly) else (if ((project observation kpi) =? (kpiUpdate)) then (leverCoalesce) else (if ((project observation kpi) =? (kpiEdit)) then (leverEditPath) else (if ((project observation kpi) =? (kpiChecked)) then (leverReuseChecks) else (if ((project observation kpi) =?
